% 绘制封面/尾页的浅色报告信息块。
% 第一个参数是蓝色块底边，第二个参数是信息块底边。
% 主体使用 SUSTBlue!10；第二个 fill 用页面底色切出一条细边，形成嵌入蓝色块的留白。
\newcommand{\SUSTDrawInfoBand}[2]{%
  \setlength{\SUSTCoverInfoBandTopY}{\dimexpr#1+\SUSTCoverInfoInsetDepth\relax}%
  \setlength{\SUSTCoverInfoBandBottomY}{#2}%
  \setlength{\SUSTCoverInfoBandHeight}{\dimexpr\SUSTCoverInfoBandTopY-\SUSTCoverInfoBandBottomY\relax}%
  \setlength{\SUSTCoverInfoInsetRun}{.5\SUSTCoverInfoInsetDepth}%
  \setlength{\SUSTCoverInfoBottomRun}{.5\SUSTCoverInfoBandHeight}%
  \fill[SUSTBlue!10]
    (0,\SUSTCoverInfoBandTopY)
    -- (\SUSTCoverInfoTopWidth,\SUSTCoverInfoBandTopY)
    -- (\dimexpr\SUSTCoverInfoTopWidth+\SUSTCoverInfoBottomRun\relax,\SUSTCoverInfoBandBottomY)
    -- (0,\SUSTCoverInfoBandBottomY)
    -- cycle;
  \fill[SUSTLightBlue]
    (0,\SUSTCoverInfoBandTopY)
    -- (\dimexpr\SUSTCoverInfoTopWidth+\SUSTCoverInfoEdgeGap\relax,\SUSTCoverInfoBandTopY)
    -- ++ (\SUSTCoverInfoInsetRun,-\SUSTCoverInfoInsetDepth)
    -- ++ (-\SUSTCoverInfoEdgeGap,0)
    -- ++ (\dimexpr-\SUSTCoverInfoInsetRun-.5\SUSTCoverInfoEdgeGap\relax,\dimexpr\SUSTCoverInfoInsetDepth-\SUSTCoverInfoEdgeGap\relax)
    -- (0,\dimexpr\SUSTCoverInfoBandTopY-\SUSTCoverInfoEdgeGap\relax)
    -- cycle;
}
